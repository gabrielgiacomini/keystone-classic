import { KeystoneField, FieldOptions } from '../index';

export interface FileFieldOptions extends FieldOptions {
  storage?: Storage;
  dest?: string;
  filename?: (item: any, file: Express.Multer.File) => string;
  path?: string;
  prefix?: string;
  datePrefix?: boolean | string;
  allowedTypes?: string[] | string;
  maxFileSize?: number;
}

export interface Storage {
  uploadFile(file: Express.Multer.File, callback: (err: Error | null, result?: any) => void): void;
  removeFile(file: any, callback: (err: Error | null) => void): void;
  getFileURL(file: any): string;
}

export interface FileField extends KeystoneField<FileFieldOptions, FileValue> {
  // File-specific methods
  validateInput(data: any, callback: (err: Error | null) => void): void;
  validateRequiredInput(item: any, data: any, callback: (err: Error | null) => void): void;
  updateItem(item: any, data: any, callback: (err: Error | null) => void): void;
  format(item: any): FileValue | null;
  getFileURL(file: FileValue): string;
}

export interface FileValue {
  filename: string;
  originalname?: string;
  path?: string;
  size?: number;
  mimetype?: string;
  url?: string;
