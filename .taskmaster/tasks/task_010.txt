# Title: Implement Error Handling Types
# Status: pending
# Dependencies: 2
# Priority: high
# Description: Create structured error interfaces for validation, database, and authorization errors to improve type safety in error handling.
# Details:
Define error handling types in `types/lib/errors.d.ts`:

```typescript
// Base Keystone error
export interface KeystoneError extends Error {
  name: string;
  message: string;
  stack?: string;
}

// Validation error
export interface ValidationError extends KeystoneError {
  name: 'ValidationError';
  errors: Record<string, FieldValidationError>;
  invalidKeys: string[];
}

export interface FieldValidationError {
  type: 'required' | 'invalid' | 'unique' | string;
  message: string;
  path: string;
  value?: any;
}

// Database error
export interface DatabaseError extends KeystoneError {
  name: 'DatabaseError';
  code?: number | string;
  collection?: string;
  index?: string;
  keyValue?: Record<string, any>;
  errmsg?: string;
}

// Authorization error
export interface AuthorizationError extends KeystoneError {
  name: 'AuthorizationError';
  message: string;
  reason?: 'not-signed-in' | 'invalid-credentials' | 'access-denied' | string;
  data?: Record<string, any>;
}

// Error creation functions
export function createValidationError(errors: Record<string, FieldValidationError>): ValidationError;
export function createDatabaseError(message: string, details?: Partial<DatabaseError>): DatabaseError;
export function createAuthorizationError(message: string, reason?: string, data?: Record<string, any>): AuthorizationError;

// Error checking functions
export function isValidationError(err: Error): err is ValidationError;
export function isDatabaseError(err: Error): err is DatabaseError;
export function isAuthorizationError(err: Error): err is AuthorizationError;

// Error handling callback types
export type ErrorCallback = (err: Error | null) => void;
export type ValidationCallback = (err: ValidationError | null) => void;
export type DatabaseCallback = (err: DatabaseError | null, result?: any) => void;
export type AuthorizationCallback = (err: AuthorizationError | null, user?: any) => void;
```

Ensure all interfaces have proper JSDoc comments explaining their purpose and usage. Include type guards for error checking and specialized callback types for different error scenarios.

# Test Strategy:
Create test cases that import and use these error types in error handling scenarios. Verify that the TypeScript compiler correctly enforces type constraints. Test with actual Keystone error handling patterns to ensure the interfaces match real-world usage.
