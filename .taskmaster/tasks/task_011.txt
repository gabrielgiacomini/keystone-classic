# Details:
Define Express integration types in `types/lib/express.d.ts`:

```typescript
import * as express from 'express';
import { Keystone } from '../index';
import { KeystoneList } from './list';

// Extend Express Request interface
declare global {
  namespace Express {
    interface Request {
      keystone?: Keystone;
      list?: KeystoneList<any>;
      user?: any;
      session?: any;
      flash?: (type: string, message: string) => void;
    }
  }
}

// Keystone middleware
export interface KeystoneMiddleware extends express.RequestHandler {
  (req: express.Request, res: express.Response, next: express.NextFunction): void;
}

// Keystone route handler
export interface KeystoneRouteHandler extends express.RequestHandler {
  (req: express.Request, res: express.Response, next: express.NextFunction): void;
}

// Admin UI middleware
export interface AdminUIMiddleware extends KeystoneMiddleware {
  (req: express.Request, res: express.Response, next: express.NextFunction): void;
}

// Session middleware options
export interface SessionOptions {
  key?: string;
  secret?: string;
  resave?: boolean;
  saveUninitialized?: boolean;
  cookie?: express.CookieOptions;
  store?: express.RequestHandler;
}

// Static middleware options
export interface StaticOptions {
  prefix?: string;
  maxAge?: number;
  includeHash?: boolean;
  configureExpress?: (app: express.Application) => void;
}

// Middleware factory functions
export function createSessionMiddleware(options: SessionOptions): express.RequestHandler;
export function createStaticMiddleware(path: string | string[], options?: StaticOptions): express.RequestHandler;
export function createBodyParserMiddleware(): express.RequestHandler[];
export function createCorsMiddleware(options?: Record<string, any>): express.RequestHandler;
```

Ensure all interfaces have proper JSDoc comments explaining their purpose and usage. Include proper extension of Express types to ensure compatibility with @types/express.
