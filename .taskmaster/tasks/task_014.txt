// Type utilities

// Make all properties in T optional
export type Partial<T> = {
  [P in keyof T]?: T[P];
};

// Make all properties in T required
export type Required<T> = {
  [P in keyof T]-?: T[P];
};

// Make all properties in T readonly
export type Readonly<T> = {
  readonly [P in keyof T]: T[P];
};

// Pick a set of properties K from T
export type Pick<T, K extends keyof T> = {
  [P in K]: T[P];
};

// Exclude properties K from T
export type Omit<T, K extends keyof T> = Pick<T, Exclude<keyof T, K>>;

// Extract keys of T that have type U
export type KeysOfType<T, U> = {
  [K in keyof T]: T[K] extends U ? K : never
}[keyof T];

// Branded type helper
export type Brand<T, B> = T & { __brand: B };

// Field path utilities

// Get the type of a field at a given path
export type FieldType<T, P extends string> = P extends keyof T
  ? T[P]
  : P extends `${infer A}.${infer B}`
  ? A extends keyof T
    ? T[A] extends Record<string, any>
      ? FieldType<T[A], B>
      : never
    : never
  : never;

// Get all possible field paths in T
export type FieldPaths<T, Prefix extends string = ''> = {
  [K in keyof T]: K extends string
    ? T[K] extends Record<string, any>
      ? `${Prefix}${K}` | `${Prefix}${K}.${FieldPaths<T[K], ''>}`
      : `${Prefix}${K}`
    : never;
}[keyof T];

// Document data utilities

// Extract document data from a Keystone document
export type KeystoneDocumentData<T> = Omit<T, '_id' | 'id' | '__v' | 'createdAt' | 'updatedAt'>;

// Create a new document type
export type NewDocument<T> = Partial<KeystoneDocumentData<T>>;

// Update a document type
export type UpdateDocument<T> = Partial<KeystoneDocumentData<T>>;

// Helper functions

// Check if a value is defined (not null or undefined)
export function isDefined<T>(value: T | null | undefined): value is T;

// Check if a value is a plain object
export function isPlainObject(value: any): value is Record<string, any>;

// Check if a value is an array
export function isArray<T>(value: any): value is T[];

// Check if a value is a string
export function isString(value: any): value is string;

// Check if a value is a number
export function isNumber(value: any): value is number;

// Check if a value is a boolean
export function isBoolean(value: any): value is boolean;

// Check if a value is a date
export function isDate(value: any): value is Date;

// Check if a value is a MongoDB ObjectId
export function isObjectId(value: any): value is mongoose.Types.ObjectId;

// Convert a value to a MongoDB ObjectId
export function toObjectId(value: string | mongoose.Types.ObjectId): mongoose.Types.ObjectId;

// Convert a value to a string
export function toString(value: any): string;

// Convert a value to a number
export function toNumber(value: any): number;

// Convert a value to a boolean
export function toBoolean(value: any): boolean;

// Convert a value to a date
export function toDate(value: any): Date;
