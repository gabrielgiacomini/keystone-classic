import * as express from 'express';
import { KeystoneList, KeystoneDocument } from './list';

// Authentication options
export interface AuthOptions {
  onFailure?: (req: express.Request, res: express.Response, next: express.NextFunction) => void;
  allowedAttempts?: number;
  lockoutTime?: number;
  lockoutExp?: number;
  failureField?: string;
  failureData?: Record<string, any>;
}

// Session options
export interface SessionOptions {
  key?: string;
  secret?: string;
  resave?: boolean;
  saveUninitialized?: boolean;
  cookie?: express.CookieOptions;
  store?: express.RequestHandler;
}

// User interface
export interface User extends KeystoneDocument {
  _id: mongoose.Types.ObjectId;
  id: string;
  name?: string;
  email?: string;
  password?: string;
  isAdmin?: boolean;
  [key: string]: any;
}

// Session interface
export interface Session {
  userId: string | mongoose.Types.ObjectId;
  startTime: Date;
  lastAccessTime: Date;
  data: Record<string, any>;
  isAuthenticated: boolean;
  user?: User;
}

// Authentication result
export interface AuthResult {
  success: boolean;
  user?: User;
  message?: string;
}

// Password reset token
export interface PasswordResetToken {
  userId: string | mongoose.Types.ObjectId;
  token: string;
  expires: Date;
}

// Authentication functions
export function authenticate(email: string, password: string, callback: (err: Error | null, result?: AuthResult) => void): void;
export function signin(user: User, req: express.Request, res: express.Response, callback: (err: Error | null) => void): void;
export function signout(req: express.Request, res: express.Response, callback: (err: Error | null) => void): void;
export function validatePassword(password: string, hash: string, callback: (err: Error | null, isValid?: boolean) => void): void;
export function hashPassword(password: string, callback: (err: Error | null, hash?: string) => void): void;
export function generateResetToken(user: User, callback: (err: Error | null, token?: PasswordResetToken) => void): void;
export function validateResetToken(token: string, callback: (err: Error | null, user?: User) => void): void;

// Middleware functions
export function requireUser(req: express.Request, res: express.Response, next: express.NextFunction): void;
export function requireAdmin(req: express.Request, res: express.Response, next: express.NextFunction): void;
export function initSessionMiddleware(options: SessionOptions): express.RequestHandler;
